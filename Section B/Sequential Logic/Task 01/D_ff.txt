library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity D_ff is
    Port (
        D     : in  STD_LOGIC;
        CLK   : in  STD_LOGIC;
        CLR   : in  STD_LOGIC;
        Q     : out STD_LOGIC;
        Q_bar : out STD_LOGIC
    );
end D_ff;

architecture Behavioral of D_ff is

    signal Q_temp : STD_LOGIC := '0';

begin

    process(CLK, CLR)
    begin
        if CLR = '0' then
            Q_temp <= '0';
        elsif rising_edge(CLK) then
            Q_temp <= D;
        end if;
    end process;

    Q     <= Q_temp;
    Q_bar <= not Q_temp;

end Behavioral;
