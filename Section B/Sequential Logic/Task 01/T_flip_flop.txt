entity T_flip_flop is
    Port (
        T     : in  STD_LOGIC;
        CLK   : in  STD_LOGIC;
        Q     : out STD_LOGIC;
        Q_bar : out STD_LOGIC
    );
end T_flip_flop;

architecture Structural of T_flip_flop is

    component D_ff
        Port (
            D     : in  STD_LOGIC;
            CLK   : in  STD_LOGIC;
            CLR   : in  STD_LOGIC;
            Q     : out STD_LOGIC;
            Q_bar : out STD_LOGIC
        );
    end component;

    component XOR_1
        Port (
            A : in  STD_LOGIC;
            B : in  STD_LOGIC;
            C : inout STD_LOGIC
        );
    end component;

    signal D_in      : STD_LOGIC;
    signal Q_int     : STD_LOGIC;
    signal Q_bar_int : STD_LOGIC;
    signal CLR_const : STD_LOGIC := '1';

begin

    -- XOR gate: D = T XOR Q
    U1 : XOR_1
        port map (
            A => T,
            B => Q_int,
            C => D_in
        );

    -- D Flip-Flop
    U2 : D_ff
        port map (
            D     => D_in,
            CLK   => CLK,
            CLR   => CLR_const,
            Q     => Q_int,
            Q_bar => Q_bar_int
        );

    -- Output connection
    Q     <= Q_int;
    Q_bar <= Q_bar_int;

end Structural;
