entity T_FF_4 is
    Port ( T : in  STD_LOGIC;
           CLK : in  STD_LOGIC;
           CLR : in  STD_LOGIC;
           Q : out  STD_LOGIC;
           Q_bar : out  STD_LOGIC);
end T_FF_4;

architecture Behavioral of T_FF_4 is

SIGNAL X: STD_LOGIC:='0';

begin
	
	process(CLK, CLR)
begin
			 if CLR = '0' then
				  X <= '0';
			 elsif rising_edge(CLK) then
				  if T = '1' then
						X <= not X;
				  else
						X <= X;
				  end if;
			 end if;
		end process;

		Q <= X;
		Q_bar <= not X;

end Behavioral;
