entity Full_Adder is
    Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           Cin : in  STD_LOGIC;
           P : inout  STD_LOGIC;
           G : inout  STD_LOGIC;
			  Q : inout  STD_LOGIC;
           Cout : out  STD_LOGIC;
			  Z : out STD_LOGIC );
			  
end Full_Adder;

architecture Behavioral of Full_Adder is

component AND_1
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : inout  STD_LOGIC);
end COMPONENT;
component AND_2
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : inout  STD_LOGIC);
end COMPONENT;


component XOR_1
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : inout  STD_LOGIC);
end COMPONENT;
component XOR_2
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : out  STD_LOGIC);
end COMPONENT;


component OR_1
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : out  STD_LOGIC);
end COMPONENT;


begin

	U1 : AND_1 PORT MAP (A,B,G);
	U2 : AND_2 PORT MAP (P,Cin,Q);
	U3 : XOR_1 PORT MAP (A,B,P);
	U4 : XOR_2 PORT MAP (P,Cin,Z);
	U5 : OR_1 PORT MAP (Q,G,Cout);

end Behavioral;
