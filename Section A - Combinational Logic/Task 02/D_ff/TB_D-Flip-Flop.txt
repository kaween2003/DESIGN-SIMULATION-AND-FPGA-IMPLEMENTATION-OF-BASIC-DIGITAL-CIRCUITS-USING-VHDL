LIBRARY ieee;
USE ieee.std_logic_1164.ALL;

ENTITY TB_D_flip_flop IS
END TB_D_flip_flop;

ARCHITECTURE behavior OF TB_D_flip_flop IS

    COMPONENT D_flip_flop
    PORT(
         D   : IN    std_logic;
         CLK : IN    std_logic;
         Q   : INOUT std_logic;
         Q1  : INOUT std_logic
        );
    END COMPONENT;

   -- Inputs
   signal D   : std_logic := '0';
   signal CLK : std_logic := '0';

   -- Outputs
   signal Q   : std_logic;
   signal Q1  : std_logic;

BEGIN

   -- Instantiate the Unit Under Test
   uut: D_flip_flop PORT MAP (
          D   => D,
          CLK => CLK,
          Q   => Q,
          Q1  => Q1
        );

   -- Stimulus process
   stim_proc: process
   begin

      -- Initialize latch with D = 0
      D <= '0';
      CLK <= '1';
      wait for 10 ns;
      CLK <= '0';
      wait for 10 ns;

      -- Test case 1: D = 1
      -- When CLK = 1, Q should become 1 and Q1 should become 0
      D <= '1';
      CLK <= '1';
      wait for 10 ns;
      CLK <= '0';
      wait for 10 ns;

      -- Test case 2: D = 0
      -- When CLK = 1, Q should become 0 and Q1 should become 1
      D <= '0';
      CLK <= '1';
      wait for 10 ns;
      CLK <= '0';
      wait for 10 ns;

      -- Test case 3: D = 1 again
      D <= '1';
      CLK <= '1';
      wait for 10 ns;
      CLK <= '0';
      wait for 10 ns;

      -- Test case 4: D = 0 again
      D <= '0';
      CLK <= '1';
      wait for 10 ns;
      CLK <= '0';
      wait for 10 ns;

      wait;
   end process;

END;LIBRARY ieee;
