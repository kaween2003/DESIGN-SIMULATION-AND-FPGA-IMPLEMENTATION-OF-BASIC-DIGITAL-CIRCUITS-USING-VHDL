entity D_flip_flop is
    Port ( D : in  STD_LOGIC;
           CLK : in  STD_LOGIC;
           Q : inout  STD_LOGIC;
           Q1 : inout  STD_LOGIC);
end D_flip_flop;

architecture Behavioral of D_flip_flop is



component NAND_1
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : inout  STD_LOGIC);
end COMPONENT;
component NAND_2
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : inout  STD_LOGIC);
end COMPONENT;
component NAND_3
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : inout  STD_LOGIC);
end COMPONENT;
component NAND_4
		Port ( A : in  STD_LOGIC;
           B : in  STD_LOGIC;
           C : inout  STD_LOGIC);
end COMPONENT;



component NOT_1
		    Port ( A : in  STD_LOGIC;
           C : inout  STD_LOGIC);
end COMPONENT;


SIGNAL S1,S2,S3 : STD_LOGIC; 
begin

	U1 : NOT_1 PORT MAP (D,S1);
	U2 : NAND_1 PORT MAP (D,CLK,S2);
	U3 : NAND_2 PORT MAP (S1,CLK,S3);
	U4 : NAND_3 PORT MAP (S2,Q1,Q);
	U5 : NAND_4 PORT MAP (Q,S3,Q1);

end Behavioral;
